-- Geneius Entertainment: corrected import (no temporary tables). -- Safe to re-run: skips bookings with the same date and private title. -- Times remain NULL because they have not been confirmed. BEGIN; INSERT INTO public.acts (name) SELECT v.name FROM (VALUES ('DJ + ABBA Mega Gems'), ('Jingle Bell Ball — DJ + Gel Duo')) AS v(name) WHERE NOT EXISTS (SELECT 1 FROM public.acts a WHERE a.name = v.name); INSERT INTO public.act_members (act_id, performer_id) SELECT a.id, p.id FROM public.acts a CROSS JOIN public.performers p WHERE a.name IN ('DJ + ABBA Mega Gems', 'Jingle Bell Ball — DJ + Gel Duo') AND p.name IN ('Gene', 'Gem', 'Megan') AND NOT EXISTS ( SELECT 1 FROM public.act_members am WHERE am.act_id = a.id AND am.performer_id = p.id ); DO $import$ DECLARE gig RECORD; chosen_act uuid; chosen_booking uuid; BEGIN FOR gig IN SELECT * FROM (VALUES ('2026-11-07','daytime','Garage / 90s Brunch — Ginger Italian','DJ GEM&I'), ('2026-11-13','evening','DJ + ABBA Mega Gems — Windsor','DJ + ABBA Mega Gems'), ('2026-11-14','evening','DJ & Karaoke — Gander Inn','The General — DJ & Karaoke'), ('2026-11-20','evening','ABBA Mega Gems — Royal Heaven','ABBA Mega Gems'), ('2026-11-27','evening','Gem Solo Singer — Royal Heaven','Gem — Solo Singer'), ('2026-11-28','evening','DJ — Gander Inn','The General — DJ'), ('2026-12-05','evening','Jingle Bell Ball — Team Social','Jingle Bell Ball — DJ + Gel Duo'), ('2026-12-11','evening','DJ — Park Grill','The General — DJ'), ('2026-12-12','evening','DJ — Gander Inn','The General — DJ'), ('2026-12-24','evening','DJ — Gander Inn','The General — DJ'), ('2026-12-26','evening','DJ — Park Grill','The General — DJ'), ('2026-12-31','evening','DJ GEM&I — Carshalton Social','DJ GEM&I') ) AS gigs(event_date, slot, title, act_name) LOOP SELECT id INTO chosen_act FROM public.acts WHERE name = gig.act_name; IF chosen_act IS NULL THEN RAISE EXCEPTION 'Missing act: %. Run starter setup.sql first.', gig.act_name; END IF; IF NOT EXISTS (SELECT 1 FROM public.act_members WHERE act_id = chosen_act) THEN RAISE EXCEPTION 'No performers linked to act: %', gig.act_name; END IF; SELECT id INTO chosen_booking FROM public.bookings WHERE event_date = gig.event_date::date AND title = gig.title LIMIT 1; IF chosen_booking IS NULL THEN INSERT INTO public.bookings (event_date, slot, title, act_id, status, start_time, end_time, notes) VALUES (gig.event_date::date, gig.slot, gig.title, chosen_act, 'confirmed', NULL, NULL, 'Imported from Geneius diary; exact times to confirm. Venue is private.') RETURNING id INTO chosen_booking; END IF; INSERT INTO public.booking_performers (booking_id, performer_id) SELECT chosen_booking, am.performer_id FROM public.act_members am WHERE am.act_id = chosen_act AND NOT EXISTS ( SELECT 1 FROM public.booking_performers bp WHERE bp.booking_id = chosen_booking AND bp.performer_id = am.performer_id ); END LOOP; END $import$; COMMIT; SELECT b.event_date, b.slot, b.title, string_agg(DISTINCT p.name, ', ' ORDER BY p.name) AS performers FROM public.bookings b JOIN public.booking_performers bp ON bp.booking_id = b.id JOIN public.performers p ON p.id = bp.performer_id WHERE b.event_date BETWEEN '2026-11-01' AND '2026-12-31' GROUP BY b.id, b.event_date, b.slot, b.title ORDER BY b.event_date, b.title;
